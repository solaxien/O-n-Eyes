cmake_minimum_required(VERSION 3.20)
project(p3_android LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SD_SOURCE "" CACHE PATH "Pinned stable-diffusion.cpp checkout")
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "")
set(SD_WEBP OFF CACHE BOOL "")
set(SD_WEBM OFF CACHE BOOL "")
set(SD_VULKAN ON CACHE BOOL "")
set(GGML_OPENMP OFF CACHE BOOL "")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")

if(NOT EXISTS "${SD_SOURCE}/include/stable-diffusion.h")
    message(FATAL_ERROR "Set SD_SOURCE to the stable-diffusion.cpp checkout")
endif()
add_subdirectory("${SD_SOURCE}" stable-diffusion)
add_executable(p3-native benchmark.cpp)
target_link_libraries(p3-native PRIVATE stable-diffusion)
target_include_directories(p3-native PRIVATE "${SD_SOURCE}/include")
